#2. 常用指令配置別名
   ## 打開用戶目錄(C:\user\{USER NAME}，創建.bashrc文件)

   touch ~/.bashrc

    - 在.bashrc文件中輸入如下內容：
   
    - 用於輸出git提交日誌
        alias lg='git log --pretty=oneline --all --graph --abbrev-commit'

    - 用於輸出當前目錄所有文件及基本信息
        alias ll='ls -al'


#3. 查詢個人 Git alias
    git config --get--regexp alias

#4. 查詢全局 Git alias
    git config --global --get--regexp alias
    
    - 設置 alias 別名
        git config --global alias.<別名> <指令>
        例：git config --global alias.st status 
    
    - My alias settings
        alias.st status
        alias.co checkout
        alias.br branch
        alias.cm commit -m
        alias.last log -1 HEAD
        alias.l5 log --oneline -n 5

#5. 遠程倉庫

    - 建立公key

        ssh-keygen -t ed25519 -C "fx53168@example.com"

    - 查看遠程

        git remote

    - 遠程操作 

        git clone git@xxxxx.xxx.x.xx (資料夾名稱可選)

	git fetch [remote name] [branch name]
            抓取(只拉不併)

	git pull [remote name] [branch name]
	    拉取(拉+併)

    	git push [--set-upstream] origin main
	    推送到遠端分支綁定關聯關係(只有第一次推送時才需要此參數)
	
	git branch -vv 查看遠程、本地關聯

    ※[ 重要 ] push 到遠端前，務必記得先pull，解決衝突後再push

#6. 刪除分支
    
    git branch -d <branch name>

    誤刪時查詢 Hash code. 例如:of3e9b
    git reflog

#7. 合併

    git merge dev --no-ff
        先checkout至被合併的分支待命, 例如：(HEAD->main) git merge dev [ 是指把dev當前的狀態合併到 main]

    git reset HEAD^ --hard
        合併錯誤時返回上一步

    git reset of3e9b --hard (先用 git reflog 查 hash code)

#8. rebase 錯誤時

    git reset ORIG_HEAD --hard
        只能回退一步
